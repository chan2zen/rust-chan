# Rust 实现缠论(缠中说禅)的通达信插件

## 使用方法

```shell
rustup target add  i686-pc-windows-msvc
cargo build --release --target i686-pc-windows-msvc
copy .\target\i686-pc-windows-msvc\release\zen_stock.dll C:\tdx\T0002\dlls
```

### 主图公式
```
LOGI:=0; {日志参数, =1打印日志}
BIZF:=0; {中枢参数，=1为笔中枢，=0为段中枢}
FRAC_X:=TDXDLL9(1, HIGH, LOW, LOGI),NODRAW;
FRAC:=IF(FRAC_X=-1 OR FRAC_X=-2 OR FRAC_X=-3, -1, IF(FRAC_X=1 OR FRAC_X=2 OR FRAC_X=3, 1, 0)),NODRAW; {笔极点}
DUAN:=IF(FRAC_X=-2, -1, IF(FRAC_X=2, 1, 0)),NODRAW; {笔极点}
TIP:=IF(FRAC_X=-3, -1, IF(FRAC_X=3, 1, 0)),NODRAW; {中阴段当前极点}

POLE_VALUE:IF(FRAC=1, HHV(H,BARSLAST(FRAC=-1)), IF(FRAC=-1, LLV(L,BARSLAST(FRAC=1)), DRAWNULL));
DRAWLINE(DUAN=-1,LLV(L,5),TIP=1, HHV(H,5),0), LINETHICK2, COLORFF8000, CIRCLEDOT;
DRAWLINE(DUAN=1,HHV(H,5),TIP=-1, LLV(L,5),0), LINETHICK2, COLORFF8000, CIRCLEDOT;

DRAWLINE(DUAN=-1 AND TIP<>-1,POLE_VALUE,DUAN=+1 AND TIP <>1,POLE_VALUE,0), LINETHICK2, COLORFF8000;
DRAWLINE(DUAN=+1 AND TIP<>1,POLE_VALUE,DUAN=-1 AND TIP <> -1,POLE_VALUE,0), LINETHICK2, COLORFF8000;

DRAWLINE(FRAC=-1, POLE_VALUE, FRAC=1, POLE_VALUE,0), LINETHICK1, COLORRED;
DRAWLINE(FRAC=1, POLE_VALUE, FRAC=-1,POLE_VALUE,0), LINETHICK1, COLORGREEN;

FLAG:=IF(BIZF=1, FRAC, DUAN);
BIZG:TDXDLL9(2, FLAG, POLE_VALUE, 101),NODRAW;
BIZD:TDXDLL9(2, FLAG, POLE_VALUE, 102),NODRAW;
BISE:TDXDLL9(2, FLAG, POLE_VALUE, 103),NODRAW;

BIZ:=(BIZG-BIZD)/2 + BIZD;

NOTEXT_BIZD:IF(BIZD>0,BIZD,DRAWNULL), COLORBROWN;
NOTEXT_BISE:STICKLINE(BISE=-2 OR BISE=2,BIZD,BIZG,1,0),LINETHICK3, COLORBROWN;
ZG:DRAWLINE(REF(BISE,1)=-2, BIZG, REFX(BISE,1)=2, BIZG, 1), COLORLIRED;
ZD:DRAWLINE(REF(BISE,1)=-2, BIZD, REFX(BISE,1)=2, BIZD, 1), COLORLIGREEN;
Z:DRAWLINE(REF(BISE,1)=-2, BIZ, REFX(BISE,1)=2, BIZ, 1), COLORYELLOW;

DRAWTEXT(FRAC=1 OR FRAC=-1, POLE_VALUE, VAR2STR(POLE_VALUE,IF(C>1.9,2,3)));
```
